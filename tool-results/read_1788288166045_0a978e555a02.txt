    39→function formatBytes(bytes: number): string {
    40→  if (!bytes) return '0 B';
    41→  const k = 1024;
    42→  const s = ['B', 'KB', 'MB', 'GB'];
    43→  const i = Math.floor(Math.log(bytes) / Math.log(k));
    44→  return parseFloat((bytes / Math.pow(k, i)).toFixed(1)) + ' ' + s[i];
    45→}
    46→function timeAgo(date: string): string {
    47→  const s = Math.floor((Date.now() - new Date(date).getTime()) / 1000);
    48→  if (s < 60) return 'just now';
    49→  if (s < 3600) return Math.floor(s / 60) + 'm ago';
    50→  if (s < 86400) return Math.floor(s / 3600) + 'h ago';
    51→  return Math.floor(s / 86400) + 'd ago';
    52→}

   354→  async function handleCreateInvite(type: 'agent' | 'human') {
   355→    if (type === 'human') setShowNewInviteHuman(true)
   356→    try {
   357→      const body: any = { type }
   358→      if (type === 'agent') {
   359→        body.role = inviteAgentRole
   360→        if (inviteExpiryHours) body.expiresIn = parseInt(inviteExpiryHours) * 3600
   361→      }
   362→      const res = await fetch(`${API}/api/teams/${selectedTeam!.slug}/invites`, {
   363→        method: 'POST',
   364→        headers: { 'Content-Type': 'application/json', Authorization: `Bearer ${selectedTeam!.ownerToken}` },
   365→        body: JSON.stringify(body),
   366→      })
   367→      const data = await res.json()
   368→      if (!res.ok) throw new Error(data.error || 'Failed')
   369→      if (type === 'human' && data.credentials) {
   370→        let creds: any = data.credentials
   371→        if (typeof creds === 'string') { try { creds = JSON.parse(creds) } catch { creds = { username: '', loginToken: creds } } }
   372→        setHumanInviteResult({
   373→          code: data.code,
   374→          credentials: typeof data.credentials === 'string' ? data.credentials : JSON.stringify(data.credentials),
   375→          username: creds.username || '',
   376→          token: creds.loginToken || creds.token || '',
   377→        })
   378→      } else if (type === 'agent') {
   379→        setInviteResult({ code: data.code, token: data.token || data.agentToken || '' })
   380→      }
   381→      setShowNewInviteAgent(false)
   382→      setShowNewInviteHuman(false)
   383→      await loadInvites(selectedTeam!.slug)
   384→    } catch (e: any) { alert(e.message); setShowNewInviteHuman(false) }
   385→  }
