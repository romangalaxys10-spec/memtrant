  1135→// ─── Pluralization helpers ──────────────────────────────────────────────────────

  1137→/** Russian-style Slavic pluralization */
  1138→function ruPlural(n: number, one: string, few: string, many: string): string {
  1139→  const abs = Math.abs(n);
  1140→  const lastTwo = abs % 100;
  1141→  const lastOne = abs % 10;
  1142→  if (lastTwo >= 11 && lastTwo <= 19) return `${n} ${many}`;
  1143→  if (lastOne === 1) return `${n} ${one}`;
  1144→  if (lastOne >= 2 && lastOne <= 4) return `${n} ${few}`;
  1145→  return `${n} ${many}`;
  1146→}
  1147→
  1148→/** Arabic pluralization (simplified: 1, 2, 3-10, 11+) */
  1149→function arPlural(n: number, one: string, two: string, few: string, many: string): string {
  1150→  if (n === 1) return one;
  1151→  if (n === 2) return two;
  1152→  if (n >= 3 && n <= 10) return few.replace('{n}', String(n));
  1153→  return many.replace('{n}', String(n));
  1154→}
  1155→
  1156→/** Hebrew pluralization (simplified: 1 singular, 2+ plural) */
  1157→function hePlural(n: number, one: string, plural: string): string {
  1158→  if (n === 1) return one;
  1159→  return `${n} ${plural}`;
  1160→}

  1236→// ─── useT — hook-like translation accessor ─────────────────────────────────────
  1237→
  1238→export function useT(lang: Lang): (key: string, replacements?: Record<string, string | number>) => string {
  1239→  return (key: string, replacements?: Record<string, string | number>): string => {
  1240→    const entry = translations[key];
  1241→    if (!entry) return key;
  1242→    let text = entry[lang] ?? entry['en'] ?? key;
  1243→    if (replacements) {
  1244→      for (const [k, v] of Object.entries(replacements)) {
  1245→        text = text.replace(new RegExp(`\\{${k}\\}`, 'g'), String(v));
  1246→      }
  1247→    }
  1248→    return text;
  1249→  };
  1250→}
