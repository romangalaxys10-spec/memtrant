    39→function formatBytes(bytes: number): string {
    40→  if (!bytes) return '0 B';
    41→  const k = 1024;
    42→  const s = ['B', 'KB', 'MB', 'GB'];
    43→  const i = Math.floor(Math.log(bytes) / Math.log(k));
    44→  return parseFloat((bytes / Math.pow(k, i)).toFixed(1)) + ' ' + s[i];
    45→}
    46→function timeAgo(date: string): string {
    47→  const s = Math.floor((Date.now() - new Date(date).getTime()) / 1000);
    48→  if (s < 60) return 'just now';
    49→  if (s < 3600) return Math.floor(s / 60) + 'm ago';
    50→  if (s < 86400) return Math.floor(s / 3600) + 'h ago';
    51→  return Math.floor(s / 86400) + 'd ago';
    52→}

   375→        headers: { 'Content-Type': 'application/json', Authorization: `Bearer ${selectedTeam!.ownerToken}` },
   376→        body: JSON.stringify(body),
   377→      })
   378→      const data = await res.json()
   379→      if (!res.ok) throw new Error(data.error || 'Failed')
   380→      if (type === 'human' && data.credentials) {
   381→        let creds: any = data.credentials
   382→        if (typeof creds === 'string') { try { creds = JSON.parse(creds) } catch { creds = { username: '', loginToken: creds } } }
   383→        setHumanInviteResult({
   384→          code: data.code,
   385→          credentials: typeof data.credentials === 'string' ? data.credentials : JSON.stringify(data.credentials),
   386→          username: creds.username || '',
   387→          token: creds.loginToken || creds.token || '',
   388→        })
   389→      } else if (type === 'agent') {
   390→        setInviteResult({ code: data.code, token: data.token || data.agentToken || '' })
   391→      }
   392→      setShowNewInviteAgent(false)
   393→      setShowNewInviteHuman(false)
