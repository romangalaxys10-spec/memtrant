    39→    39→function formatBytes(bytes: number): string {
    40→    40→  if (!bytes) return '0 B';
    41→    41→  const k = 1024;
    42→    42→  const s = ['B', 'KB', 'MB', 'GB'];
    43→    43→  const i = Math.floor(Math.log(bytes) / Math.log(k));
    44→    44→  return parseFloat((bytes / Math.pow(k, i)).toFixed(1)) + ' ' + s[i];
    45→    45→}
    46→    46→function timeAgo(date: string): string {
    47→    47→  const s = Math.floor((Date.now() - new Date(date).getTime()) / 1000);
    48→    48→  if (s < 60) return 'just now';
    49→    49→  if (s < 3600) return Math.floor(s / 60) + 'm ago';
    50→    50→  if (s < 86400) return Math.floor(s / 3600) + 'h ago';
    51→    51→  return Math.floor(s / 86400) + 'd ago';
    52→    52→}
