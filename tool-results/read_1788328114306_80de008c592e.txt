    40→function formatBytes(bytes: number): string {
    41→  if (!bytes) return '0 B';
    42→  const k = 1024;
    43→  const s = ['B', 'KB', 'MB', 'GB'];
    44→  const i = Math.floor(Math.log(bytes) / Math.log(k));
    45→  return parseFloat((bytes / Math.pow(k, i)).toFixed(1)) + ' ' + s[i];
    46→}

   158→  function formatTimeAgo(date: string): string {
   159→    const s = Math.floor((Date.now() - new Date(date).getTime()) / 1000)
   160→    if (s < 60) return timeAgo.justNow
   161→    if (s < 3600) return timeAgo.minutesAgo(Math.floor(s / 60))
   162→    if (s < 86400) return timeAgo.hoursAgo(Math.floor(s / 3600))
   163→    return timeAgo.daysAgo(Math.floor(s / 86400))
   164→  }

   387→      }

   389→        method: 'POST',

   392→      })
   393→      const data = await res.json()
   394→      if (!res.ok) throw new Error(data.error || 'Failed')
   395→      if (type === 'human' && data.credentials) {
   396→        let creds: any = data.credentials
   397→        if (typeof creds === 'string') { try { creds = JSON.parse(creds) } catch { creds = { username: '', loginToken: creds } } }
   398→        setHumanInviteResult({
   399→          code: data.code,
   400→          credentials: typeof data.credentials === 'string' ? data.credentials : JSON.stringify(data.credentials),
   401→          username: creds.username || '',
   402→          token: creds.loginToken || creds.token || '',
   403→        })
   404→      } else if (type === 'agent') {
   405→        setInviteResult({ code: data.code, token: data.token || data.agentToken || '' })
   406→      }
   407→      setShowNewInviteAgent(false)
   408→      setShowNewInviteHuman(false)
